cmake_minimum_required(VERSION 3.25)
project(Ludot VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LUAU_BUILD_WEB OFF CACHE BOOL "" FORCE)
add_subdirectory(luau)
option(LUDOT_ASSERTIONS "Keep Luau compiler invariants checked" ON)
if(LUDOT_ASSERTIONS)
    target_compile_definitions(Luau.Compiler PRIVATE LUAU_ENABLE_ASSERT)
    target_compile_definitions(Luau.Ast PRIVATE LUAU_ENABLE_ASSERT)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME ludot_extensions COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_extensions.py ${CMAKE_BINARY_DIR}/luau)
add_test(NAME ludot_language COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_language.py ${CMAKE_BINARY_DIR}/luau)
add_library(Ludot.Language STATIC language/src/shader.cpp)
target_include_directories(Ludot.Language PUBLIC language/include)
target_link_libraries(Ludot.Language PUBLIC Luau.Ast)
add_executable(ludot-shader tools/shader_main.cpp)
target_link_libraries(ludot-shader PRIVATE Ludot.Language Luau.Config)
add_test(NAME ludot_shaders COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_shaders.py $<TARGET_FILE:ludot-shader>)
add_test(NAME ludot_lsp COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_lsp.py)
add_test(NAME ludot_documentation COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_documentation.py)
install(TARGETS ludot-shader Luau.Repl.CLI Luau.Analyze.CLI Luau.Compile.CLI RUNTIME DESTINATION bin)
install(DIRECTORY tools/ludot_lsp DESTINATION lib/ludot FILES_MATCHING PATTERN "*.py")

install(PROGRAMS tools/ludot-lsp DESTINATION bin)

install(FILES types/core.d.luau types/engine.api DESTINATION share/ludot)
